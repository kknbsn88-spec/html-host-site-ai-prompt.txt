# DATABASE STRUCTURE

  [TABLE: sites]
    id          INTEGER PRIMARY KEY AUTOINCREMENT
    site_name   TEXT NOT NULL DEFAULT ''
    slug        TEXT NOT NULL UNIQUE
    created_at  TEXT DEFAULT (datetime('now'))

  [TABLE: files]
    id          INTEGER PRIMARY KEY AUTOINCREMENT
    site_id     INTEGER NOT NULL REFERENCES sites(id) ON DELETE CASCADE
    filename    TEXT NOT NULL
    file_size   INTEGER DEFAULT 0
    mime_type   TEXT DEFAULT ''
    created_at  TEXT DEFAULT (datetime('now'))

  [TABLE: settings]
    key         TEXT PRIMARY KEY

# ROUTING SYSTEM

  .htaccess rules for Apache:
    /              -> index.php
    /login         -> login.php
    /logout        -> logout.php
    /dashboard     -> dashboard.php
    /upload        -> upload.php

    /settings      -> settings.php
    /ht/{slug}/{file} -> serve.php?slug={slug}&file={file}

  router.php for PHP built-in server (php -S localhost:8080 router.php):
    Handles same routes as .htaccess
    Also serves /assets/ files directly
    Returns 301 redirect for /ht/{slug}/ to /ht/{slug}/index.html

# SERVE SYSTEM — serve.php

  Accepts : GET ?slug=X&file=Y  or clean URL /ht/{slug}/{file}
  Security:
    Sanitize slug — allow only alphanumeric, hyphen, underscore
    Sanitize filename — basename() only, strip ../ and // sequences
    Reject if file does not exist or is not a file
  MIME map covers:
    html htm css js json xml txt pdf
    jpg jpeg png gif svg ico webp bmp
    mp3 wav ogg aac flac
    mp4 webm ogv
    woff woff2 ttf eot
    zip
  Headers:
    Content-Type from MIME map, fallback application/octet-stream
    Content-Length from filesize()
    Cache-Control: public, max-age=3600
  Fallback: 404 response with clean styled HTML error page matching brand

# JAVASCRIPT SYSTEM — app.js

  Module 1 — Particle Engine:
    On DOMContentLoaded, create 25 div.particle elements and append to body
    Each gets random: width 3-8px, height same, left position 0-100%,
    animation-duration 8-20s, animation-delay 0-15s, opacity 0.3-0.7
    CSS handles the float-up keyframe, JS only creates and randomizes them

  Module 2 — Modal Manager:
    openModal(id) — adds .open class, locks body scroll, focuses first input
    closeModal(id) — removes .open class, restores scroll
    Click outside modal box closes it
    Escape key closes active modal
    Applied to: create site modal on dashboard

  Module 3 — Drag and Drop:
    For every .upload-zone element:
      dragenter and dragover — add .drag-over class (purple glow border)
      dragleave and drop — remove .drag-over class
      drop — assign event.dataTransfer.files to the hidden file input
    fileInput change event — calls renderFileList(files)
    renderFileList — builds preview rows: SVG icon + filename + size + X remove button
    X button — splices file from FileList-like array, re-renders list

  Module 4 — Live Search:
    Input event on #search field
    Queries all .site-card elements
    Filters by data-name and data-slug attributes (both lowercase)
    Hides non-matching cards with display:none
    Shows .empty-search element if zero cards visible

  Module 5 — Clipboard:
    copyToClipboard(text, buttonEl) function
    Uses navigator.clipboard.writeText()
    On success: swaps button inner SVG to checkmark SVG for 2000ms, then reverts
    On failure: falls back to document.execCommand copy

  Module 6 — Count Up:
    countUp(el, target, duration) function using requestAnimationFrame
    easeOutCubic easing curve
    IntersectionObserver watches all .count-up elements
    Triggers once on first intersection, reads data-target attribute for value

  Module 7 — Upload Progress:
    On upload form submit:
    Disables submit button, swaps text to "Uploading..." with spinner SVG
    Shows .progress-bar-wrap, animates bar width with setInterval to 85%
    Clears interval on page navigation

  Module 8 — Alerts:
    Auto-dismiss .alert elements after 4000ms with fade-out transition
    Slide-in animation on appear
